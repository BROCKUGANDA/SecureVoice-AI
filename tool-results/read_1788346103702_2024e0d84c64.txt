    29→const AGENT_DUR = 7.5; // virtual seconds agent "speaks" per line

    32→function eventDur(e: ScenarioEvent) {
    33→  if (e.speaker === "agent") return AGENT_DUR + e.en.length / 40;
    34→  if (e.speaker === "customer") return CUSTOMER_DUR;
    35→  return 2.2;
    36→}
